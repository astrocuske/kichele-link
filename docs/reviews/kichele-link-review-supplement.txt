===== supabase/migrations/002_lock_down_rls.sql =====
-- 002_lock_down_rls.sql
-- Hardens applications, payment_transactions, and profiles for client access:
--   applications:
--     - clients can only INSERT their own application, always status 'pending',
--       only on open gigs, with a bounded message
--     - posters can only UPDATE pending -> accepted/rejected; contact columns
--       are immutable for clients
--   payment_transactions:
--     - client INSERTs must satisfy: payer owns the gig, amount equals the
--       gig's agreed price (whole KES, 1..1_000_000), the payee has an
--       accepted application, and status is forced to 'pending'
--   profiles:
--     - phone/location are no longer readable by other users; names are
--       served through the public_profile view, own row through RPCs
--   grants:
--     - anon loses all direct table access

begin;

-- ============================================================
-- Helper: role of the current PostgREST/Supabase request
-- ============================================================
create or replace function public.request_role()
returns text
language sql
stable
set search_path = ''
as $$
    select coalesce(current_setting('request.jwt.claim.role', true), 'anon');
$$;

-- ============================================================
-- applications integrity
-- ============================================================
create or replace function public.enforce_application_rules()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
    caller_role text := public.request_role();
begin
    if caller_role = 'service_role' then
        return new;
    end if;

    if tg_op = 'INSERT' then
        -- Workers never choose their own status: applications start pending.
        new.worker_id := auth.uid();
        new.status := 'pending';

        if not exists (
            select 1 from public.gigs g
            where g.id = new.gig_id
              and g.status = 'open'
        ) then
            raise exception 'Applications can only be submitted to open gigs.'
            using errcode = '23514';
        end if;

        if new.message is not null and length(new.message) > 2000 then
            raise exception 'Application message must be 2000 characters or fewer.'
            using errcode = '22001';
        end if;
    end if;

    if tg_op = 'UPDATE' then
        -- Only a status transition is allowed, and only from pending.
        if old.gig_id is distinct from new.gig_id
            or old.worker_id is distinct from new.worker_id
            or old.message is distinct from new.message
            or old.created_at is distinct from new.created_at
        then
            raise exception 'Only the application status can be updated.'
            using errcode = '23514';
        end if;

        if old.status is distinct from 'pending' then
            raise exception 'Only pending applications can be updated.'
            using errcode = '23514';
        end if;

        if new.status not in ('accepted', 'rejected') then
            raise exception 'Applications can only be set to accepted or rejected.'
            using errcode = '23514';
        end if;
    end if;

    return new;
end;
$$;

drop trigger if exists applications_enforce_rules on public.applications;

create trigger applications_enforce_rules
before insert or update on public.applications
for each row
execute function public.enforce_application_rules();

-- ============================================================
-- payment_transactions integrity
-- ============================================================
create or replace function public.enforce_payment_transaction_rules()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
    caller_role text := public.request_role();
    gig public.gigs;
begin
    if caller_role = 'service_role' then
        return new;
    end if;

    if tg_op = 'INSERT' or tg_op = 'UPDATE' then
        new.status := 'pending';
    end if;

    if tg_op = 'INSERT' then
        select * into gig
        from public.gigs g
        where g.id = new.gig_id;

        if gig.id is null then
            raise exception 'Payment does not reference an existing gig.'
            using errcode = '23514';
        end if;

        if gig.poster_id is distinct from new.payer_id then
            raise exception 'Payment records can only be created by the gig poster.'
            using errcode = '23514';
        end if;

        if gig.agreed_price is null or gig.agreed_price is distinct from new.amount then
            raise exception 'Payment amount must equal the agreed price on the gig.'
            using errcode = '23514';
        end if;

        if new.amount is null
            or new.amount <> floor(new.amount)
            or new.amount < 1
            or new.amount > 1000000
        then
            raise exception 'Payment amount must be a whole number between KES 1 and KES 1,000,000.'
            using errcode = '23514';
        end if;

        if not exists (
            select 1 from public.applications a
            where a.gig_id = new.gig_id
              and a.worker_id = new.worker_id
              and a.status = 'accepted'
        ) then
            raise exception 'The selected worker does not have an accepted application for this gig.'
            using errcode = '23514';
        end if;

        new.payer_id := gig.poster_id;
    end if;

    return new;
end;
$$;

drop trigger if exists payment_transactions_enforce_rules on public.payment_transactions;

create trigger payment_transactions_enforce_rules
before insert or update on public.payment_transactions
for each row
execute function public.enforce_payment_transaction_rules();

-- ============================================================
-- profiles: hide phone/location from other users
-- ============================================================
create or replace view public.public_profile
with (security_barrier = true)
as
    select p.user_id, p.full_name
    from public.profiles p;

revoke all on public.public_profile from anon;
grant select on public.public_profile to authenticated;

create or replace function public.get_my_profile()
returns table (
    full_name text,
    phone text,
    location text
)
language sql
security definer
set search_path = public
as $$
    select p.full_name, p.phone, p.location
    from public.profiles p
    where p.user_id = auth.uid();
$$;

create or replace function public.upsert_my_profile(
    p_full_name text,
    p_phone text,
    p_location text
)
returns table (
    full_name text,
    phone text,
    location text
)
language plpgsql
security definer
set search_path = public
as $$
begin
    p_full_name := btrim(coalesce(p_full_name, ''));
    p_phone := btrim(coalesce(p_phone, ''));
    p_location := btrim(coalesce(p_location, ''));

    if p_full_name = '' or length(p_full_name) > 100 then
        raise exception 'Full name is required and must be 100 characters or fewer.'
        using errcode = '23514';
    end if;

    if length(p_phone) > 20 then
        raise exception 'Phone number must be 20 characters or fewer.'
        using errcode = '22001';
    end if;

    if length(p_location) > 100 then
        raise exception 'Location must be 100 characters or fewer.'
        using errcode = '22001';
    end if;

    insert into public.profiles (user_id, full_name, phone, location)
    values (
        auth.uid(),
        nullif(p_full_name, ''),
        nullif(p_phone, ''),
        nullif(p_location, '')
    )
    on conflict (user_id) do update
    set full_name = excluded.full_name,
        phone = excluded.phone,
        location = excluded.location
    where profiles.user_id = auth.uid();

    return query
    select p.full_name, p.phone, p.location
    from public.profiles p
    where p.user_id = auth.uid();
end;
$$;

-- Clients no longer read/write the base table directly.
drop policy if exists "Authenticated users can view profile names" on public.profiles;
drop policy if exists "Users can view their own profile" on public.profiles;
drop policy if exists "Enable insert for users based on user_id" on public.profiles;

revoke all on public.profiles from anon;
revoke all on public.profiles from authenticated;

-- ============================================================
-- Grant hygiene: every direct-table write is grant-scoped
-- ============================================================
revoke all on public.gigs from anon;
revoke all on public.applications from anon;
revoke all on public.payment_transactions from anon;
revoke all on sequence public.gigs_id_seq
    , public.applications_id_seq
    , public.payment_transactions_id_seq
    , public.profiles_id_seq
from anon;

grant select, insert, update on public.applications to authenticated;
grant select, insert on public.payment_transactions to authenticated;

commit;

		id: number;
		title: string;

		created_at: string;
		poster_id: string;

	let loading = $state(true);
	let submitting = $state(false);
	let errorMessage = $state('');

		try {
			const {
				data: { user }
			} = await supabase.auth.getUser();

			if (!user) {
				await goto('/login');
				return;
			}

			const gigId = Number(window.location.pathname.split('/').pop());

			if (!gigId) {

					'id, title, description, location, budget, status, created_at, poster_id'

			if (error) {
				errorMessage = error.message;
				return;
			}

			gig = data;
		} catch (error) {
			errorMessage =
				error instanceof Error ? error.message : 'Something went wrong.';
		} finally {
			loading = false;
		}
	});

	async function handleApply() {
		errorMessage = '';
		successMessage = '';
		submitting = true;

		try {
			const {
				data: { user }
			} = await supabase.auth.getUser();

			if (!user || !gig) {
				errorMessage = 'You must be logged in to apply.';
				return;
			}

			if (gig.poster_id === user.id) {
				errorMessage = 'You cannot apply to your own gig.';
				return;
			}

			const { error } = await supabase.from('applications').insert({
				gig_id: gig.id,
				worker_id: user.id,
				message: message || null,
				status: 'pending'
			});

			if (error) {
				if (error.code === '23505') {
					errorMessage = 'You have already applied for this gig.';
				} else {
					errorMessage = error.message;
				}
				return;
			}

			successMessage = 'Application submitted successfully!';
			message = '';
		} catch (error) {
			errorMessage =
				error instanceof Error ? error.message : 'Something went wrong.';
		} finally {
			submitting = false;
		}
	}

<script lang="ts">
	import { onMount } from 'svelte';
	import { goto } from '$app/navigation';
	import { supabase } from '#lib/supabase/client.ts';

	const categories = [
		'Cleaning',
		'Construction',
		'Delivery',
		'Gardening',
		'Hair & Beauty',
		'Moving',
		'Repairs',
		'Transport',
		'Other'
	];

	let loading = $state(true);
	let saving = $state(false);
	let errorMessage = $state('');
	let successMessage = $state('');

	let gigId = $state<number | null>(null);

	let title = $state('');
	let description = $state('');
	let category = $state('');
	let location = $state('');
	let budget = $state('');

	onMount(async () => {
		try {
			const {
				data: { user },
				error: authError
			} = await supabase.auth.getUser();

			if (authError || !user) {
				goto('/login');
				return;
			}

			// URL format: /gigs/5/edit
			// The gig ID is at index 2.
			const id = Number(window.location.pathname.split('/')[2]);

			if (!id || Number.isNaN(id)) {
				errorMessage = 'Invalid gig.';
				return;
			}

			gigId = id;

			const { data: gig, error: gigError } = await supabase
				.from('gigs')
				.select(
					'id, title, description, category, location, budget, poster_id'
				)
				.eq('id', id)
				.single();

			if (gigError) {
				errorMessage = gigError.message;
				return;
			}

			if (gig.poster_id !== user.id) {
				errorMessage = 'You can only edit your own gigs.';
				return;
			}

			title = gig.title;
			description = gig.description;
			category = gig.category ?? '';
			location = gig.location ?? '';
			budget = gig.budget?.toString() ?? '';
		} catch (error) {
			errorMessage =
				error instanceof Error ? error.message : 'Something went wrong.';
		} finally {
			loading = false;
		}
	});

	async function saveGig() {
		errorMessage = '';
		successMessage = '';

		if (!gigId) {
			errorMessage = 'Invalid gig.';
			return;
		}

		if (!title.trim()) {
			errorMessage = 'Please enter a gig title.';
			return;
		}

		if (!description.trim()) {
			errorMessage = 'Please enter a gig description.';
			return;
		}

		saving = true;

		const { error } = await supabase
			.from('gigs')
			.update({
				title: title.trim(),
				description: description.trim(),
				category: category || null,
				location: location.trim() || null,
				budget: budget ? Number(budget) : null
			})
			.eq('id', gigId);

		if (error) {
			errorMessage = error.message;
			saving = false;
			return;
		}

		successMessage = 'Gig updated successfully.';
		saving = false;
	}

	function goBack() {
		goto('/dashboard');
	}
</script>

	async function loadGigs() {

			.from('gigs')
			.select(
				'id, created_at, poster_id, title, description, location, budget, status, category, agreed_price, payment_status'
			)
			.eq('status', 'open')
			.order('created_at', { ascending: false });

		let profileMap = new Map<string, string>();

		if (posterIds.length > 0) {
			const { data: profiles } = await supabase
				.from('public_profile')
				.select('user_id, full_name')
				.in('user_id', posterIds);

			profileMap = new Map(
				(profiles ?? [])
					.filter((profile) => profile.user_id)
					.map((profile) => [profile.user_id as string, profile.full_name || 'Kichele Link user'])
			);
		}

		gigs = loadedGigs.map((gig) => ({
			...gig,
			poster_name: profileMap.get(gig.poster_id) || 'Kichele Link user'
		}));

		loading = false;
	}

	function filteredGigs() {
		const search = searchTerm.trim().toLowerCase();

		return gigs.filter((gig) => {

	import { supabase } from '#lib/supabase/client.ts';

	let userId = $state('');
	let email = $state('');
	let fullName = $state('');
	let phone = $state('');
	let location = $state('');

	let loading = $state(true);
	let saving = $state(false);
	let message = $state('');
	let error = $state('');

	type MyProfile = {
		full_name: string | null;
		phone: string | null;
		location: string | null;
	};

	onMount(async () => {
		try {
			const {
				data: { user },
				error: authError
			} = await supabase.auth.getUser();

			if (authError || !user) {
				goto('/login');
				return;
			}

			userId = user.id;
			email = user.email ?? '';

			const { data: profileData, error: profileError } = await supabase.rpc(
				'get_my_profile'
			);

			if (profileError) {
				console.error('Profile load failed:', profileError);
				error = `${profileError.message} (Code: ${profileError.code ?? 'unknown'})`;
				return;
			}

			const profile = (profileData as MyProfile[] | null)?.[0] ?? null;

			if (profile) {
				fullName = profile.full_name ?? '';
				phone = profile.phone ?? '';
				location = profile.location ?? '';
			} else {
				fullName = user.user_metadata?.full_name ?? '';
			}
		} catch (loadError) {
			console.error('Profile loading error:', loadError);
			error = 'Something went wrong while loading your profile.';
		} finally {
			loading = false;
		}
	});

	async function saveProfile(event: SubmitEvent) {
		event.preventDefault();

		if (saving) return;

		error = '';
		message = '';

		const cleanedName = fullName.trim();
		const cleanedPhone = phone.trim();
		const cleanedLocation = location.trim();

		if (!cleanedName) {
			error = 'Please enter your full name.';
			return;
		}

		if (cleanedName.length > 100) {
			error = 'Your name must be 100 characters or fewer.';
			return;
		}

		if (cleanedPhone.length > 20) {
			error = 'Your phone number must be 20 characters or fewer.';
			return;
		}

		if (cleanedLocation.length > 100) {
			error = 'Your location must be 100 characters or fewer.';
			return;
		}

		if (!userId) {
			error = 'Your session has expired. Please sign in again.';
			goto('/login');
			return;
		}

		saving = true;

		try {
			const { error: saveError } = await supabase.rpc('upsert_my_profile', {
				p_full_name: cleanedName,
				p_phone: cleanedPhone,
				p_location: cleanedLocation
			});

			if (saveError) {
				console.error('Profile save failed:', saveError);

				error = `${saveError.message} (Code: ${saveError.code ?? 'unknown'})`;
				return;
			}

			fullName = cleanedName;
			phone = cleanedPhone;
			location = cleanedLocation;

			message = 'Your profile has been saved successfully.';
		} catch (saveError) {
			console.error('Profile saving error:', saveError);

			error = 'Something went wrong while saving your profile.';
		} finally {
			saving = false;
		}
	}
